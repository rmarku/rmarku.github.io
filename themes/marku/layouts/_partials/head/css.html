{{- /* Builds one stylesheet from the self-hosted @font-face rules plus the theme's own styles, and preloads the body face. The old pipeline minified unconditionally and had no source maps at all. */ -}} {{- $fonts := resources.Get "css/fonts.css" | resources.ExecuteAsTemplate "css/fonts.css" . }} {{- /* Still Sass until the design-system rewrite lands; same options either way. */ -}} {{- $opts := dict "transpiler" "dartsass" "targetPath" "css/theme.css" "enableSourceMap" (not hugo.IsProduction) "outputStyle" (cond hugo.IsProduction "compressed" "expanded") }} {{- $theme := resources.Get "scss/main.scss" | css.Sass $opts }} {{- $css := slice $fonts $theme | resources.Concat "css/style.css" }} {{- if hugo.IsProduction }} {{- $css = $css | minify | fingerprint }} {{- else }} {{- end }} {{- /* The body face is needed for first paint; the others can arrive late. */ -}} {{- with resources.Get "fonts/source-serif-4-latin-wght-normal.woff2" }} {{- end }}
